# Form Builder

Generate entire forms from a plain array of field definitions. The form builder handles layout, validation, submission, and reactive state.

## `useFormBuilder()`

```ts
import { useFormBuilder } from '~/composables/useFormBuilder'
import { z } from 'zod'

const schema = z.object({
  firstName: z.string().min(1, 'Required'),
  lastName: z.string().min(1, 'Required'),
  email: z.string().email('Invalid email'),
  role: z.enum(['admin', 'member', 'viewer']),
  bio: z.string().max(500).optional(),
  newsletter: z.boolean(),
})

const {
  fields,
  values,
  errors,
  isValid,
  submitting,
  submitted,
  dirty,
  handleSubmit,
  validate,
  validateField,
  setFieldValue,
  setFieldError,
  reset,
  formContext,
} = useFormBuilder({
  fields: [
    {
      name: 'firstName',
      type: 'text',
      label: 'First name',
      placeholder: 'Jane',
      required: true,
      colSpan: 1,
    },
    {
      name: 'lastName',
      type: 'text',
      label: 'Last name',
      placeholder: 'Doe',
      required: true,
      colSpan: 1,
    },
    {
      name: 'email',
      type: 'email',
      label: 'Email',
      placeholder: 'jane@co.io',
      required: true,
      colSpan: 2,
    },
    {
      name: 'role',
      type: 'select',
      label: 'Role',
      colSpan: 1,
      options: [
        { label: 'Admin', value: 'admin' },
        { label: 'Member', value: 'member' },
        { label: 'Viewer', value: 'viewer' },
      ],
    },
    {
      name: 'bio',
      type: 'textarea',
      label: 'Bio',
      placeholder: 'Tell us about yourself…',
      colSpan: 2,
    },
    { name: 'newsletter', type: 'checkbox', label: 'Subscribe to newsletter', colSpan: 2 },
  ],
  schema,
  initialValues: { role: 'member', newsletter: false },
  onSubmit: async (data) => {
    await api.createUser(data)
  },
})
```

### Options

| Option          | Type                                   | Description                           |
| --------------- | -------------------------------------- | ------------------------------------- |
| `fields`        | `FormFieldDefinition[]`                | Ordered array of field definitions    |
| `schema`        | `StandardSchemaV1`                     | Optional form-level validation schema |
| `initialValues` | `Partial<T>`                           | Override the default blank values     |
| `onSubmit`      | `(values: T) => void \| Promise<void>` | Called after successful validation    |

### Return values

| Key             | Type                                  | Description                                    |
| --------------- | ------------------------------------- | ---------------------------------------------- |
| `fields`        | `ComputedRef<FormBuilderField[]>`     | Fields enriched with a reactive `error` string |
| `values`        | `T` (reactive)                        | Live two-way-bindable form values              |
| `errors`        | `Record<string, string \| undefined>` | Field-keyed error map                          |
| `isValid`       | `ComputedRef<boolean>`                | `true` when all errors are cleared             |
| `submitting`    | `ComputedRef<boolean>`                | `true` while submit handler runs               |
| `submitted`     | `ComputedRef<boolean>`                | `true` after first submit attempt              |
| `dirty`         | `ComputedRef<boolean>`                | Any value changed from initial                 |
| `handleSubmit`  | `() => Promise<void>`                 | Validates then calls `onSubmit`                |
| `validate`      | `() => Promise<boolean>`              | Validates all fields, returns `true` if valid  |
| `validateField` | `(name) => Promise<boolean>`          | Validates a single field                       |
| `setFieldValue` | `(name, value) => void`               | Programmatically set a field value             |
| `setFieldError` | `(name, msg) => void`                 | Inject a server-side error message             |
| `reset`         | `(values?) => void`                   | Resets to initial state                        |
| `formContext`   | `FormContext`                         | Underlying Formwerk context for advanced use   |

## `UiFormBuilder` component

Renders a `useFormBuilder` result into a responsive grid with a submit button.

```html
<template>
  <UiFormBuilder
    :fields="fields"
    :values="values"
    :submitting="submitting"
    submit-label="Create user"
    :cols="2"
    @submit="handleSubmit"
    @update:field="(name, val) => setFieldValue(name, val)"
  >
    <template #actions="{ submitting }">
      <div class="flex gap-3 justify-end">
        <button type="button" @click="reset">Cancel</button>
        <button type="submit" :disabled="submitting">Save</button>
      </div>
    </template>
  </UiFormBuilder>
</template>
```

### Props

| Prop          | Type                      | Default      | Description                       |
| ------------- | ------------------------- | ------------ | --------------------------------- |
| `fields`      | `FormBuilderField[]`      | **required** | From `useFormBuilder().fields`    |
| `values`      | `Record<string, unknown>` | **required** | From `useFormBuilder().values`    |
| `submitting`  | `boolean`                 | `false`      | Disables submit and shows spinner |
| `submitLabel` | `string`                  | `'Submit'`   | Submit button text                |
| `cols`        | `1 \| 2 \| 3`             | `1`          | Grid column count (responsive)    |

## `FormFieldDefinition` type

```ts
interface FormFieldDefinition extends BaseFieldProps {
  name: string // form value path — supports dots: "address.city"
  type: FormFieldType // one of the 35 types
  placeholder?: string
  hint?: string // helper text below the field
  colSpan?: 1 | 2 | 3 // grid column span inside UiFormBuilder

  // Selection fields (select, radio, combobox, autocomplete, toggle-group, tree-select)
  options?: { label: string; value: string; disabled?: boolean }[]
}
```

## `UiFormField` — dynamic rendering

`UiFormField` renders any field component by reading the `type` from a `FormFieldDefinition`. `UiFormBuilder` uses it internally, but you can also use it directly.

```html
<!-- Single dynamic field -->
<UiFormField
  :field="{ name: 'email', type: 'email', label: 'Work email', placeholder: 'you@company.com', required: true }"
  :model-value="values.email"
  @update:model-value="values.email = $event"
/>

<!-- Loop over a mixed field list -->
<UiFormField
  v-for="f in fields"
  :key="f.name"
  :field="f"
  :model-value="values[f.name]"
  @update:model-value="setFieldValue(f.name, $event)"
/>
```

## Validation

### Supported schema libraries

Vuestrata uses the [`@standard-schema/spec`](https://github.com/standard-schema/standard-schema) interface. Any compliant library works:

::code-group

```ts [Zod]
import { z } from 'zod'

const schema = z.object({
  username: z
    .string()
    .min(3)
    .max(20)
    .regex(/^[a-z0-9_]+$/),
  email: z.string().email(),
  age: z.number().int().min(18, 'You must be 18 or older'),
  plan: z.enum(['free', 'pro', 'team']),
  agree: z.literal(true, { message: 'You must accept the terms' }),
})
```

```ts [Valibot]
import * as v from 'valibot'

const schema = v.object({
  username: v.pipe(v.string(), v.minLength(3), v.maxLength(20)),
  email: v.pipe(v.string(), v.email()),
  age: v.pipe(v.number(), v.integer(), v.minValue(18)),
  plan: v.picklist(['free', 'pro', 'team']),
  agree: v.literal(true),
})
```

```ts [Yup]
import * as yup from 'yup'

const schema = yup.object({
  username: yup.string().min(3).max(20).required(),
  email: yup.string().email().required(),
  age: yup.number().integer().min(18).required(),
  plan: yup.mixed().oneOf(['free', 'pro', 'team']).required(),
  agree: yup.boolean().oneOf([true], 'Required'),
})
```

```ts [ArkType]
import { type } from 'arktype'

const schema = type({
  username: 'string >= 3 & string <= 20',
  email: 'string.email',
  age: 'number.integer >= 18',
  plan: '"free" | "pro" | "team"',
})
```

### Field-level schemas

Attach a schema directly to a field for self-contained validation:

```html
<UiTextField
  name="slug"
  label="URL slug"
  placeholder="my-page-slug"
  :schema="slugSchema"
  hint="Only lowercase letters, numbers, and hyphens"
/>

<script setup lang="ts">
  import { z } from 'zod'
  const slugSchema = z
    .string()
    .min(3, 'At least 3 characters')
    .max(60, 'Too long')
    .regex(/^[a-z0-9-]+$/, 'Only lowercase letters, numbers, and hyphens')
</script>
```

### Cross-field validation

Use Zod's `.refine()` to validate fields against each other:

```ts
const changePasswordSchema = z
  .object({
    current: z.string().min(8),
    next: z.string().min(8),
    confirm: z.string(),
  })
  .refine((d) => d.next === d.confirm, {
    message: 'Passwords do not match',
    path: ['confirm'],
  })
```

### Injecting server errors

Push API error messages onto specific fields:

```ts
const { setFieldError, handleSubmit } = useFormBuilder({ fields, schema })

async function submit(values) {
  try {
    await api.register(values)
  } catch (err) {
    if (err.code === 'EMAIL_TAKEN') {
      setFieldError('email', 'This email is already registered')
    }
  }
}
```

## Async submission

The `onSubmit` callback can be `async`. `submitting` remains `true` until the promise settles.

```ts
const { handleSubmit, submitting } = useFormBuilder({
  fields,
  schema,
  onSubmit: async (values) => {
    const response = await api.post('/users', values)
    router.push(`/users/${response.id}`)
  },
})
```

```html
<button type="submit" :disabled="submitting" @click="handleSubmit">
  {{ submitting ? 'Saving…' : 'Save' }}
</button>
```

## Advanced patterns

### Conditional fields

Show or hide fields based on sibling values:

```html
<UiForm :initial-values="{ type: 'individual', companyName: '' }">
  <template #default="{ values }">
    <UiRadioGroup
      name="type"
      label="Account type"
      :options="[
        { label: 'Individual', value: 'individual' },
        { label: 'Business',   value: 'business' },
      ]"
    />

    <UiTextField
      v-if="values.type === 'business'"
      name="companyName"
      label="Company name"
      placeholder="Acme Inc."
      required
    />
  </template>
</UiForm>
```

### Dependent validation

Compute a schema that reacts to another field's value:

```ts
const { values } = useFormBuilder({ fields, schema })

const phoneSchema = computed(() =>
  values.country === 'US'
    ? z.string().regex(/^\d{10}$/, '10-digit US number required')
    : z.string().min(7, 'Enter a valid phone number').optional(),
)
```

### Programmatic control

```ts
const { setFieldValue, setFieldError, validate, reset } = useFormBuilder({ fields, schema })

// Apply a promo code
async function applyPromo(code: string) {
  const discount = await api.validatePromo(code)
  if (discount) {
    setFieldValue('discount', discount)
  } else {
    setFieldError('promoCode', 'Invalid or expired promo code')
  }
}

// Preview without submitting
async function preview() {
  const ok = await validate()
  if (ok) showPreview(values)
}

// Reset on navigation
onBeforeRouteLeave(() => reset())
```

### Using `formContext` directly

`formContext` is the raw Formwerk `FormContext`. Use it for custom field composables or third-party integrations:

```ts
const { formContext } = useFormBuilder({ fields })
provide('vuestrata:form', formContext)

// In child component
const formCtx = inject('vuestrata:form')
const { fieldValue, setValue, errorMessage } = useCustomField({
  name: 'signature',
  label: 'Signature',
  context: formCtx,
})
```
